cmake_minimum_required(VERSION 3.30)
project(AdventureCraftNativeRendering LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(JNI REQUIRED)
find_package(Vulkan REQUIRED)
find_package(Threads REQUIRED)
if(NOT AC_NATIVE_SDK_ROOT)
    message(FATAL_ERROR "Run the accepted installNativeRenderingSdks Gradle task first")
endif()
# All upstream dependencies are pinned and verified by the Gradle installer.
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
foreach(dependency IN ITEMS SHADERMAKE MATHLIB NRI VULKAN_HEADERS VMA NGX)
    string(TOLOWER "${dependency}" directory)
    if(dependency STREQUAL "VULKAN_HEADERS")
        set(directory "vulkan-headers")
    elseif(dependency STREQUAL "NGX")
        set(directory "dlss")
    endif()
    set(FETCHCONTENT_SOURCE_DIR_${dependency} "${AC_NATIVE_SDK_ROOT}/${directory}" CACHE PATH "" FORCE)
endforeach()
set(SHADERMAKE_FIND_COMPILERS OFF CACHE BOOL "" FORCE)
set(SHADERMAKE_FIND_DXC OFF CACHE BOOL "" FORCE)
set(SHADERMAKE_FIND_DXC_VK OFF CACHE BOOL "" FORCE)
if(WIN32)
    set(AC_DXC "${AC_NATIVE_SDK_ROOT}/dxc/bin/x64/dxc.exe")
else()
    set(AC_DXC "${AC_NATIVE_SDK_ROOT}/dxc/bin/dxc")
endif()
set(SHADERMAKE_DXC_PATH "${AC_DXC}" CACHE FILEPATH "" FORCE)
set(SHADERMAKE_DXC_VK_PATH "${AC_DXC}" CACHE FILEPATH "" FORCE)
set(NRD_STATIC_LIBRARY ON CACHE BOOL "" FORCE)
set(NRD_NRI ON CACHE BOOL "" FORCE)
set(NRD_NORMAL_ENCODING 4 CACHE STRING "" FORCE)
set(NRD_ROUGHNESS_ENCODING 1 CACHE STRING "" FORCE)
set(NRD_EMBEDS_SPIRV_SHADERS ON CACHE BOOL "" FORCE)
set(NRD_EMBEDS_DXIL_SHADERS OFF CACHE BOOL "" FORCE)
set(NRD_EMBEDS_DXBC_SHADERS OFF CACHE BOOL "" FORCE)
set(NRI_STATIC_LIBRARY ON CACHE BOOL "" FORCE)
set(NRI_ENABLE_NVTX_SUPPORT OFF CACHE BOOL "" FORCE)
set(NRI_ENABLE_NGX_SDK OFF CACHE BOOL "" FORCE)
set(NRI_ENABLE_D3D11_SUPPORT OFF CACHE BOOL "" FORCE)
set(NRI_ENABLE_D3D12_SUPPORT OFF CACHE BOOL "" FORCE)
set(NRI_ENABLE_FFX_SDK OFF CACHE BOOL "" FORCE)
set(NRI_ENABLE_NIS_SDK OFF CACHE BOOL "" FORCE)
set(NRI_ENABLE_XESS_SDK OFF CACHE BOOL "" FORCE)
add_subdirectory("${AC_NATIVE_SDK_ROOT}/nrd" nrd)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    # GCC 16 diagnoses upstream's allocator/destructor templates under -O3.
    # Keep warnings visible without changing the verified SDK sources.
    target_compile_options(NRI_VK PRIVATE -Wno-error=maybe-uninitialized -Wno-error=uninitialized)
    target_compile_options(NRI_Validation PRIVATE -Wno-error=maybe-uninitialized -Wno-error=uninitialized)
endif()
add_subdirectory(fsr)
add_library(ac_render_native SHARED NativeBridge.cpp ngx/NgxBackend.cpp ngx/NgxFrameGeneration.cpp NrdBackend.cpp)
target_include_directories(ac_render_native PRIVATE ${JNI_INCLUDE_DIRS} "${AC_NATIVE_SDK_ROOT}/dlss/include")
if(WIN32)
    set(AC_NGX_LIBRARY "${AC_NATIVE_SDK_ROOT}/dlss/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib")
else()
    set(AC_NGX_LIBRARY "${AC_NATIVE_SDK_ROOT}/dlss/lib/Linux_x86_64/libnvsdk_ngx.a")
endif()
target_link_libraries(ac_render_native PRIVATE NRD NRDIntegration NRI ac_fsr Vulkan::Vulkan "${AC_NGX_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
target_compile_definitions(ac_render_native PRIVATE NRD_NORMAL_ENCODING=4 NRD_ROUGHNESS_ENCODING=1)
if(UNIX AND NOT APPLE)
    target_link_options(ac_render_native PRIVATE -Wl,--no-undefined)
endif()
